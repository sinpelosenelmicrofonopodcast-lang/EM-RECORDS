cmake_minimum_required(VERSION 3.22)
project(MixMaestro VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.12 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(MixMaestro COMPANY_NAME "EM Records" BUNDLE_ID com.emrecords.mixmaestro
 PLUGIN_MANUFACTURER_CODE Emrc PLUGIN_CODE Mm01 FORMATS VST3
 PRODUCT_NAME "Mix Maestro" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(MixMaestro PRIVATE Source/Plugin.cpp)
target_compile_definitions(MixMaestro PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(MixMaestro PRIVATE juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
juce_add_console_app(MixMaestroTests PRODUCT_NAME "Mix Maestro Tests")
target_sources(MixMaestroTests PRIVATE tests/plugin_test.cpp)
target_compile_definitions(MixMaestroTests PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(MixMaestroTests PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_recommended_config_flags)
